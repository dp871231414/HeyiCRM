常用命令

改完前端代码需要重新构建，8000 才会用上新版本

# 启动（也可以直接在访达里双击 启动看板.command）
./启动看板.command

# 前端构建 → 同步给后端同源托管（让 8000 用上新版本）
cd frontend && npm run build
rm -rf ../backend/static/assets && cp -R dist/assets ../backend/static/assets
cp dist/index.html ../backend/static/index.html

# 后端测试（应为 81 passed）
cd backend && .venv/bin/python -m pytest -q

# 发布到线上（也可在 WorkBuddy 的「设置—数据管理—应用」里管理）
# 部署目录：backend　启动命令：sh start.sh

# 停止全部服务
pkill -f "uvicorn app.main:app"; pkill -f "vite"
